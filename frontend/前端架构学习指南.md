#### 路由守卫

```typescript
router.beforeEach(async (to, from, next) => {
  // 1. 启动进度条
  NProgress.start()

  // 2. 设置页面标题
  if (to.meta.title) {
    document.title = `${to.meta.title} - TradingAgents-CN`
  }

  // 3. 认证检查
  const authStore = useAuthStore()
  if (to.meta.requiresAuth && !authStore.isAuthenticated) {
    // 保存原始路径,登录后跳转
    authStore.setRedirectPath(to.fullPath)
    next('/login')
    return
  }

  // 4. 已登录用户访问登录页,重定向到仪表板
  if (authStore.isAuthenticated && to.name === 'Login') {
    next('/dashboard')
    return
  }

  // 5. 更新当前路由信息
  const appStore = useAppStore()
  appStore.setCurrentRoute(to)

  next()
})
```

## 6. API 调用机制

### 6.1 Axios 封装 (`src/api/request.ts`)

#### 统一 API 响应格式

```typescript
interface ApiResponse<T = any> {
  success: boolean    // 业务状态码
  data: T            // 响应数据
  message: string    // 响应消息
  code?: number      // 业务错误码
  timestamp?: string // 时间戳
  request_id?: string // 请求 ID
}
```

#### ApiClient 使用示例

```typescript
import { ApiClient } from '@/api/request'

// GET 请求
const response = await ApiClient.get<UserInfo>('/api/auth/me')

const loginResponse = await ApiClient.post<LoginResponse>(
```

### 6.2 请求拦截器

#### 自动添加请求头

```typescript
instance.interceptors.request.use((config) => {
  const authStore = useAuthStore()
  const appStore = useAppStore()

  // 1. 添加认证头
  if (!config.skipAuth && authStore.token) {
    config.headers.Authorization = `Bearer ${authStore.token}`
  }

  // 2. 添加请求 ID (用于日志追踪)
  config.headers['X-Request-ID'] = generateRequestId()

  // 3. 添加语言头
  config.headers['Accept-Language'] = appStore.language

  // 4. 显示加载状态
  if (config.showLoading) {
    appStore.setLoading(true)
  }

  return config
})
```

### 6.3 响应拦截器

#### 统一错误处理

```typescript
instance.interceptors.response.use(
  (response) => {
    // 1. 隐藏加载状态
    if (response.config.showLoading) {
      appStore.setLoading(false)
    }

    // 2. 检查业务状态码
    const data = response.data as ApiResponse
    if (!data.success) {
      // 处理业务错误
      handleBusinessError(data)
      return Promise.reject(new Error(data.message))
    }

    // 3. 返回 response.data
    return response.data
  },
  async (error) => {
    // 1. 隐藏加载状态
    if (error.config?.showLoading) {
      appStore.setLoading(false)
    }

    // 2. 处理 HTTP 错误
    if (error.response) {
      switch (error.response.status) {
        case 401:
          // 尝试刷新 Token
          if (!error.config.skipAuth && authStore.refreshToken) {
            const success = await authStore.refreshAccessToken()
            if (success) {
              // 重试原请求
              return instance.request(error.config)
            }
          }
          // 跳转登录页
          handle401Error(authStore)
          break

        case 403:
          showErrorMessage('权限不足')
          break

        case 500:
          showErrorMessage('服务器错误')
          break
      }
    }

    // 3. 网络错误处理
    if (error.code === 'ECONNABORTED') {
      // 尝试重试
      if (await shouldRetry(error.config, error)) {
        return retryRequest(instance, error.config)
      }
      showErrorMessage('请求超时')
    }

    return Promise.reject(error)
  }
)
```

```typescript
  state: (): AuthState => ({
    isAuthenticated: !!localStorage.getItem('auth-token'),
    token: localStorage.getItem('auth-token'),
    refreshToken: localStorage.getItem('refresh-token'),
    user: JSON.parse(localStorage.getItem('user-info') || 'null'),

    roles: [],

    redirectPath: '/'
  }),

  getters: {
    // 用户显示名称

      return this.user?.username || this.user?.email || '未知用户'
    },

    // 是否为管理员
    isAdmin(): boolean {
      return this.roles.includes('admin')
    },

    // 用户统计信息
    userStats(): Record<string, number> {
      return {
        totalAnalyses: this.user?.total_analyses || 0,
        dailyQuota: this.user?.daily_quota || 1000
      }
    }
  },

  actions: {
    // 登录
    async login(loginForm: LoginForm) {
      this.loginLoading = true
      try {
        const response = await authApi.login(loginForm)
        if (response.success) {
          const { access_token, refresh_token, user } = response.data

          // 保存认证信息
          this.setAuthInfo(access_token, refresh_token, user)

          // 开源版 admin 拥有所有权限
          this.permissions = ['*']
          this.roles = ['admin']

          // 启动 Token 自动刷新
          setupTokenRefreshTimer()

          return true
        }
        return false
      } finally {
        this.loginLoading = false
      }
    },

    // 登出
    async logout() {
      try {
        await authApi.logout()
      } finally {
        // 清除本地认证信息
        this.clearAuthInfo()
        this.redirectToLogin()
      }
    },

    // 刷新 Token
    async refreshAccessToken() {
      if (!this.refreshToken) {
        throw new Error('没有刷新令牌')
      }

      const response = await authApi.refreshToken(this.refreshToken)
      if (response.success) {
        const { access_token, refresh_token } = response.data
        this.setAuthInfo(access_token, refresh_token)
        return true
      }

      // 刷新失败,清除认证信息
      this.clearAuthInfo()
      this.redirectToLogin()
      return false
    }
  }
})
```

```typescript
export function setupTokenRefreshTimer() {
  const authStore = useAuthStore()

  // 每 60 秒检查一次 Token 状态
  const intervalId = setInterval(async () => {
    if (!authStore.token) {
      clearInterval(intervalId)
      return
    }

    // 检查 Token 是否即将过期 (提前 5 分钟刷新)

      console.log('🔄 Token 即将过期,自动刷新...')
      await authStore.refreshAccessToken()
    }
  }, 60 * 1000) // 60 秒

  return intervalId
}
```

```typescript
function isTokenExpiringSoon(token: string, seconds: number): boolean {
  try {
    // 解析 JWT Token
    const payload = JSON.parse(atob(token.split('.')[1]))
    const exp = payload.exp * 1000 // 转换为毫秒
    const now = Date.now()

    // 检查是否在指定秒数后过期
    return (exp - now) < seconds * 1000
  } catch {
    return false
  }
}
```

```typescript
class WebSocketClient {
  private ws: WebSocket | null = null
  private reconnectTimer: number | null = null
  private url: string

  constructor(url: string) {
    this.url = url
  }

  connect() {
    this.ws = new WebSocket(this.url)

    this.ws.onopen = () => {
      console.log('✅ WebSocket 连接成功')
    }

    this.ws.onmessage = (event) => {
      const data = JSON.parse(event.data)
      this.handleMessage(data)
    }

    this.ws.onerror = (error) => {
      console.error('❌ WebSocket 错误:', error)
    }

    this.ws.onclose = () => {
      console.log('🔌 WebSocket 连接关闭')
      // 自动重连
      this.reconnect()
    }
  }

  reconnect() {
    if (this.reconnectTimer) return

    this.reconnectTimer = window.setTimeout(() => {
      console.log('🔄 WebSocket 重连中...')
      this.connect()
      this.reconnectTimer = null
    }, 5000)
  }

  handleMessage(data: any) {
    // 处理服务端推送的消息
    switch (data.type) {
      case 'TASK_UPDATE':
        // 更新任务状态
        break
      case 'SYNC_STATUS':
        // 更新同步状态
        break
    }
  }

  send(data: any) {
    if (this.ws?.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify(data))
    }
  }

  close() {
    if (this.reconnectTimer) {
      clearTimeout(this.reconnectTimer)
    }
    this.ws?.close()
  }
}
```

### 10.3 创建新 Store

#### 步骤 1: 定义 Store

```typescript
// src/stores/myFeature.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { myFeatureApi } from '@/api/myFeature'

export interface MyData {
  id: string
  name: string
  value: number
}

export const useMyFeatureStore = defineStore('myFeature', () => {
  // State
  const dataList = ref<MyData[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)

  // Getters
  const totalCount = computed(() => dataList.value.length)
  const totalValue = computed(() =>
    dataList.value.reduce((sum, item) => sum + item.value, 0)
  )

  // Actions
  async function fetchData() {
    loading.value = true
    error.value = null

    try {
      const response = await myFeatureApi.getData()
      if (response.success) {
        dataList.value = response.data
      }
    } catch (err: any) {
      error.value = err.message
    } finally {
      loading.value = false
    }
  }

  async function createItem(data: Omit<MyData, 'id'>) {
    loading.value = true
    error.value = null

    try {
      const response = await myFeatureApi.createData(data)
      if (response.success) {
        dataList.value.push(response.data)
        return true
      }
      return false
    } catch (err: any) {
      error.value = err.message
      return false
    } finally {
      loading.value = false
    }
  }

  async function deleteItem(id: string) {
    loading.value = true
    error.value = null

    try {
      await myFeatureApi.deleteData(id)
      dataList.value = dataList.value.filter(item => item.id !== id)
      return true
    } catch (err: any) {
      error.value = err.message
      return false
    } finally {
      loading.value = false
    }
  }

  return {
    // State
    dataList,
    loading,
    error,

    // Getters
    totalCount,
    totalValue,

    // Actions
    fetchData,
    createItem,
    deleteItem
  }
})
```

#### 步骤 2: 在组件中使用 Store

```vue

      </li>
    </ul>
    <el-button @click="handleCreate">创建新项目</el-button>
  </div>
</template>
```
