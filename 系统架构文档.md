## 系统概述

### 系统定位

TradingAgents-CN 是一个**基于多智能体和大模型的股票分析平台**，采用前后端分离架构：

- **核心功能**: AI 驱动的股票分析、多数据源整合、实时行情展示
- **技术特点**: FastAPI 后端 + Vue 3 前端 + MongoDB/Redis 双数据库
- **应用场景**: 股票研究、投资决策辅助、AI 分析学习

### 核心启动代码

**文件**: `app/main.py` (app/main.py:15-72)

```python
from fastapi import FastAPI
from contextlib import asynccontextmanager

@asynccontextmanager
async def lifespan(app: FastAPI):
    """应用生命周期管理"""
    # ========== 启动阶段 ==========
    # 1. 设置日志
    setup_logging()
    logger = logging.getLogger("app.main")

    # 2. 验证配置
    from app.core.startup_validator import validate_startup_config
    validate_startup_config()

    # 3. 初始化数据库
    await init_db()

    # 4. 配置桥接（将配置写入环境变量）
    from app.core.config_bridge import bridge_config_to_env
    bridge_config_to_env()

    # 5. 应用动态配置
    from app.services.config_provider import provider as config_provider
    eff = await config_provider.get_effective_system_settings()
    setup_logging(log_level=eff.get("log_level", "INFO"))

    # 6. 启动后台任务调度器
    from app.services.scheduler_service import get_scheduler_service
    scheduler_service = get_scheduler_service()
    await scheduler_service.start()

    # 7. 启动行情接收服务
    from app.services.quotes_ingestion_service import QuotesIngestionService
    quotes_service = QuotesIngestionService()
    await quotes_service.start()

    logger.info("✅ 系统启动完成")

    yield  # 应用运行中

    # ========== 关闭阶段 ==========
    # 关闭数据库连接
    await close_db()

    # 停止后台任务
    await scheduler_service.stop()
    await quotes_service.stop()

    logger.info("✅ 系统已关闭")


# 创建 FastAPI 应用
app = FastAPI(
    title="TradingAgents-CN API",
    description="多智能体股票分析平台",
    version=get_version(),
    lifespan=lifespan  # 注册生命周期管理
)
```

```python
# app/services/my_feature_service.py

class MyFeatureService:
    """我的功能服务"""

    def __init__(self):
        self.db = get_database()

    async def create_item(self, item: ItemCreate) -> Item:
        """创建项目"""
        # 业务逻辑
        item_dict = item.model_dump()
        item_dict["created_at"] = datetime.utcnow()
        result = await self.db.items.insert_one(item_dict)
        item_dict["id"] = str(result.inserted_id)
        return Item(**item_dict)

    async def get_items(self, skip: int = 0, limit: int = 100) -> list[Item]:
        """获取项目列表"""
        cursor = self.db.items.find().skip(skip).limit(limit)
        items = await cursor.to_list(length=limit)
        return [Item(**item) for item in items]
```

```python
async def get_item(item_id: str) -> Item:
    # 先查缓存
    cached = await redis.get(f"item:{item_id}")
    if cached:
        return Item.parse_raw(cached)

    # 缓存未命中，查数据库
    item = await db.items.find_one({"_id": ObjectId(item_id)})

    await redis.setex(f"item:{item_id}", 3600, item.json())
```
