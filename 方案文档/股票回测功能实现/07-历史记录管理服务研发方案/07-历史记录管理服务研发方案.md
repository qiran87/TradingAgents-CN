# 07-历史记录管理服务研发方案

## 一、领域服务概述

### 1.1 服务定位

**历史记录管理服务**负责回测历史记录的保存、查询、对比和删除，帮助用户管理自己的回测历史，方便回顾和对比不同策略的回测结果。

### 1.2 业务边界

**本服务负责**：
- 回测结果保存为历史记录
- 历史记录查询（支持多条件筛选）
- 历史记录详情查看
- 历史记录对比（最多3条）
- 历史记录删除与回收

**本服务不负责**：
- 回测结果计算（由结果计算服务负责）
- 回测执行（由回测执行引擎负责）

### 1.3 核心能力

1. **历史记录保存**
   - 保存回测结果到历史记录
   - 记录名称自定义
   - 记录描述添加

2. **历史记录查询**
   - 列表查询（分页）
   - 条件筛选（按策略、股票代码、日期范围）
   - 关键词搜索（按名称、描述）

3. **历史记录详情**
   - 查看完整回测结果
   - 查看交易明细
   - 查看资金曲线

4. **历史记录对比**
   - 选择2-3条记录对比
   - 并列展示关键指标
   - 对比资金曲线

5. **历史记录删除**
   - 删除历史记录
   - 回收站功能（可选）

### 2.1 数据库设计

#### MongoDB集合：backtest_history

**集合名称**: `backtest_history`

**索引设计**:
```javascript
db.backtest_history.createIndex({ record_id: 1 }, { unique: true })
db.backtest_history.createIndex({ user_id: 1, created_at: -1 })
db.backtest_history.createIndex({ strategy_id: 1 })
db.backtest_history.createIndex({ stock_code: 1 })
```

**数据结构**:
```typescript
interface BacktestHistoryDocument {
  _id: ObjectId
  record_id: string                // 历史记录唯一ID
  user_id: string                  // 用户ID

  // 基本信息
  name: string                     // 记录名称
  description?: string             // 记录描述
  tags?: string[]                  // 标签

  // 回测参数
  parameters: {
    stock_code: string
    start_date: string
    end_date: string
    initial_capital: number
    strategy_id: string
    strategy_params: object
  }

  // 回测结果（引用backtest_results）
  results_id: string               // 关联的backtest_results._id

  // 关键指标快照（便于列表展示）
  metrics_snapshot: {
    total_return: number
    max_drawdown: number
    sharpe_ratio: number
    win_rate: number
    total_trades: number
  }

  created_at: DateTime
  updated_at: DateTime
}
```

### 2.2 API路由

```python
@router.post("/backtest/{backtest_id}/save")
async def save_to_history(
    backtest_id: str,
    request: SaveToHistoryRequest,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """保存回测结果到历史记录"""
    # 1. 获取回测结果
    result = await db.backtest_results.find_one({"backtest_id": backtest_id})
    if not result:
        raise HTTPException(status_code=404, detail="Results not found")

    # 2. 生成记录ID
    record_id = f"hist_{datetime.utcnow().strftime('%Y%m%d_%H%M%S_%f')}"

    # 3. 创建历史记录
    history_doc = {
        "record_id": record_id,
        "user_id": "default",
        "name": request.name,
        "description": request.description,
        "tags": request.tags or [],
        "parameters": result["parameters"],
        "results_id": str(result["_id"]),
        "metrics_snapshot": {
            "total_return": result["return_metrics"]["total_return"],
            "max_drawdown": result["risk_metrics"]["max_drawdown"],
            "sharpe_ratio": result["risk_adjusted_metrics"]["sharpe_ratio"],
            "win_rate": result["trading_stats"]["win_rate"],
            "total_trades": result["trading_stats"]["total_trades"]
        },
        "created_at": datetime.utcnow(),
        "updated_at": datetime.utcnow()
    }

    await db.backtest_history.insert_one(history_doc)

    return {"record_id": record_id, "message": "Saved to history"}

@router.get("/backtest/history")
async def get_history_list(
    skip: int = 0,
    limit: int = 20,
    strategy_id: Optional[str] = None,
    stock_code: Optional[str] = None,
    search: Optional[str] = None,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """获取历史记录列表"""
    query = {"user_id": "default"}

    if strategy_id:
        query["parameters.strategy_id"] = strategy_id

    if stock_code:
        query["parameters.stock_code"] = stock_code

    if search:
        query["$text"] = {"$search": search}

    cursor = db.backtest_history.find(query).sort("created_at", -1).skip(skip).limit(limit)
    records = await cursor.to_list(length=limit)

    total = await db.backtest_history.count_documents(query)

    return {"records": records, "total": total}

@router.get("/backtest/history/{record_id}")
async def get_history_detail(
    record_id: str,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """获取历史记录详情"""
    history = await db.backtest_history.find_one({"record_id": record_id})

    if not history:
        raise HTTPException(status_code=404, detail="History record not found")

    # 获取完整结果
    results = await db.backtest_results.find_one({"_id": ObjectId(history["results_id"])})

    return {
        "history": history,
        "results": results
    }

@router.post("/backtest/history/compare")
async def compare_history(
    request: CompareHistoryRequest,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """对比历史记录"""
    if len(request.record_ids) > 3:
        raise HTTPException(status_code=400, detail="Can compare up to 3 records")

    records = []
    for record_id in request.record_ids:
        history = await db.backtest_history.find_one({"record_id": record_id})
        if not history:
            continue

        results = await db.backtest_results.find_one({"_id": ObjectId(history["results_id"])})

        records.append({
            "record_id": record_id,
            "name": history["name"],
            "metrics_snapshot": history["metrics_snapshot"],
            "results": results
        })

    return {"records": records}

@router.delete("/backtest/history/{record_id}")
async def delete_history(
    record_id: str,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """删除历史记录"""
    result = await db.backtest_history.delete_one({"record_id": record_id})

    if result.deleted_count == 0:
        raise HTTPException(status_code=404, detail="History record not found")

    return {"message": "Deleted successfully"}
```
