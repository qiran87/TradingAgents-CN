# 08-导出服务研发方案

## 一、领域服务概述

### 1.1 服务定位

**导出服务**负责将回测结果导出为Excel文件或图表图片，方便用户保存、分享和进一步分析。

### 1.2 业务边界

**本服务负责**：
- Excel文件生成（包含完整回测结果、交易明细、资金曲线）
- 图表PNG导出（前端生成，后端提供数据支持）
- 导出文件管理（临时存储、下载链接）
- 批量导出（可选）

**本服务不负责**：
- 图表渲染（由前端ECharts负责）
- 图表截图（由前端canvas-to-blob负责）

### 1.3 核心能力

1. **Excel导出**
   - 生成包含多个工作表的Excel文件
   - 工作表：摘要、详细指标、交易明细、资金曲线
   - 格式化样式（标题、加粗、边框等）
   - 文件下载

2. **图表导出**
   - 前端ECharts生成图表
   - 导出为PNG图片
   - 支持高分辨率导出

3. **文件管理**
   - 临时文件存储
   - 生成下载链接
   - 定时清理过期文件

### 2.1 Excel生成实现

```python
import openpyxl
from openpyxl.styles import Font, Alignment, PatternFill, Border, Side
from openpyxl.utils import get_column_letter
from typing import Dict, List
import io

class ExcelExporter:
    """Excel导出器"""

    def export_backtest_results(
        self,
        backtest_id: str,
        history_record: Dict,
        results: Dict,
        trades: List[Dict],
        equity_curve: Dict
    ) -> bytes:
        """导出完整回测结果到Excel"""
        wb = openpyxl.Workbook()

        # 删除默认工作表
        wb.remove(wb.active)

        # 1. 摘要工作表
        self._create_summary_sheet(wb, history_record, results)

        # 2. 详细指标工作表
        self._create_metrics_sheet(wb, results)

        # 3. 交易明细工作表
        self._create_trades_sheet(wb, trades)

        # 4. 资金曲线工作表
        self._create_equity_curve_sheet(wb, equity_curve)

        # 保存到内存
        output = io.BytesIO()
        wb.save(output)
        output.seek(0)

        return output.read()

    def _create_summary_sheet(self, wb, history_record, results):
        """创建摘要工作表"""
        ws = wb.create_sheet("回测摘要")

        # 标题
        ws['A1'] = "回测结果摘要"
        ws['A1'].font = Font(size=16, bold=True)
        ws.merge_cells('A1:D1')

        # 基本信息
        row = 3
        ws[f'A{row}'] = "记录名称"
        ws[f'B{row}'] = history_record.get("name", "")
        row += 1

        ws[f'A{row}'] = "股票代码"
        ws[f'B{row}'] = history_record["parameters"]["stock_code"]
        row += 1

        ws[f'A{row}'] = "策略"
        ws[f'B{row}'] = history_record["parameters"]["strategy_id"]
        row += 1

        ws[f'A{row}'] = "回测期间"
        ws[f'B{row}'] = f"{history_record['parameters']['start_date']} ~ {history_record['parameters']['end_date']}"
        row += 1

        ws[f'A{row}'] = "初始资金"
        ws[f'B{row}'] = f"¥{history_record['parameters']['initial_capital']:,.2f}"
        row += 2

        # 关键指标
        ws[f'A{row}'] = "关键指标"
        ws[f'A{row}'].font = Font(size=14, bold=True)
        row += 1

        metrics = results["metrics_snapshot"]

        ws[f'A{row}'] = "总收益率"
        ws[f'B{row}'] = f"{metrics['total_return']*100:.2f}%"
        row += 1

        ws[f'A{row}'] = "最大回撤"
        ws[f'B{row}'] = f"{metrics['max_drawdown']*100:.2f}%"
        row += 1

        ws[f'A{row}'] = "夏普比率"
        ws[f'B{row}'] = f"{metrics['sharpe_ratio']:.4f}"
        row += 1

        ws[f'A{row}'] = "胜率"
        ws[f'B{row}'] = f"{metrics['win_rate']*100:.2f}%"
        row += 1

        ws[f'A{row}'] = "交易次数"
        ws[f'B{row}'] = str(metrics['total_trades'])

        # 设置列宽
        ws.column_dimensions['A'].width = 20
        ws.column_dimensions['B'].width = 30

    def _create_metrics_sheet(self, wb, results):
        """创建详细指标工作表"""
        ws = wb.create_sheet("详细指标")

        # 收益指标
        row = 1
        ws[f'A{row}'] = "收益指标"
        ws[f'A{row}'].font = Font(size=14, bold=True)
        row += 2

        return_metrics = results["return_metrics"]
        ws[f'A{row}'] = "总收益率"
        ws[f'B{row}'] = f"{return_metrics['total_return']*100:.4f}%"
        row += 1

        ws[f'A{row}'] = "年化收益率"
        ws[f'B{row}'] = f"{return_metrics['annual_return']*100:.4f}%"
        row += 2

        # 风险指标
        ws[f'A{row}'] = "风险指标"
        ws[f'A{row}'].font = Font(size=14, bold=True)
        row += 2

        risk_metrics = results["risk_metrics"]
        ws[f'A{row}'] = "最大回撤"
        ws[f'B{row}'] = f"{risk_metrics['max_drawdown']*100:.4f}%"
        row += 1

        ws[f'A{row}'] = "波动率"
        ws[f'B{row}'] = f"{risk_metrics['volatility']*100:.4f}%"
        row += 1

        ws[f'A{row}'] = "下行波动率"
        ws[f'B{row}'] = f"{risk_metrics['downside_volatility']*100:.4f}%"
        row += 1

        ws[f'A{row}'] = "VaR (95%)"
        ws[f'B{row}'] = f"{risk_metrics['var_95']*100:.4f}%"
        row += 2

        # 风险调整收益指标
        ws[f'A{row}'] = "风险调整收益指标"
        ws[f'A{row}'].font = Font(size=14, bold=True)
        row += 2

        ra_metrics = results["risk_adjusted_metrics"]
        ws[f'A{row}'] = "夏普比率"
        ws[f'B{row}'] = f"{ra_metrics['sharpe_ratio']:.4f}"
        row += 1

        ws[f'A{row}'] = "索提诺比率"
        ws[f'B{row}'] = f"{ra_metrics['sortino_ratio']:.4f}"
        row += 1

        ws[f'A{row}'] = "卡玛比率"
        ws[f'B{row}'] = f"{ra_metrics['calmar_ratio']:.4f}"

    def _create_trades_sheet(self, wb, trades):
        """创建交易明细工作表"""
        ws = wb.create_sheet("交易明细")

        # 表头
        headers = ["日期", "类型", "价格", "股数", "金额", "佣金", "印花税", "总费用", "现金后", "持仓后"]
        for col, header in enumerate(headers, 1):
            cell = ws.cell(row=1, column=col)
            cell.value = header
            cell.font = Font(bold=True)
            cell.fill = PatternFill(start_color="DDDDDD", end_color="DDDDDD", fill_type="solid")

        # 数据
        for row, trade in enumerate(trades, 2):
            ws.cell(row=row, column=1).value = trade["date"]
            ws.cell(row=row, column=2).value = "买入" if trade["trade_type"] == "buy" else "卖出"
            ws.cell(row=row, column=3).value = trade["price"]
            ws.cell(row=row, column=4).value = trade["shares"]
            ws.cell(row=row, column=5).value = trade["amount"]
            ws.cell(row=row, column=6).value = trade["commission"]
            ws.cell(row=row, column=7).value = trade["stamp_duty"]
            ws.cell(row=row, column=8).value = trade["total_cost"]
            ws.cell(row=row, column=9).value = trade["cash_after"]
            ws.cell(row=row, column=10).value = trade["position_after"]

        # 设置列宽
        for col in range(1, 11):
            ws.column_dimensions[get_column_letter(col)].width = 15

    def _create_equity_curve_sheet(self, wb, equity_curve):
        """创建资金曲线工作表"""
        ws = wb.create_sheet("资金曲线")

        # 表头
        headers = ["日期", "总资产", "现金", "持仓市值"]
        for col, header in enumerate(headers, 1):
            cell = ws.cell(row=1, column=col)
            cell.value = header
            cell.font = Font(bold=True)

        # 数据
        for row, (date, assets, cash, position) in enumerate(
            zip(
                equity_curve["dates"],
                equity_curve["total_assets"],
                equity_curve["cash"],
                equity_curve["position_value"]
            ), 2
        ):
            ws.cell(row=row, column=1).value = date
            ws.cell(row=row, column=2).value = assets
            ws.cell(row=row, column=3).value = cash
            ws.cell(row=row, column=4).value = position
```

### 2.2 API接口

```python
@router.get("/backtest/{backtest_id}/export/excel")
async def export_excel(
    backtest_id: str,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """导出Excel文件"""
    # 获取历史记录（如果有）
    history = await db.backtest_history.find_one({"backtest_id": backtest_id})

    # 获取结果
    results = await db.backtest_results.find_one({"backtest_id": backtest_id})

    if not results:
        raise HTTPException(status_code=404, detail="Results not found")

    # 获取交易明细
    trades_cursor = db.backtest_trades.find({"backtest_id": backtest_id}).sort("date", 1)
    trades = await trades_cursor.to_list(length=1000)

    # 生成Excel
    exporter = ExcelExporter()
    excel_data = exporter.export_backtest_results(
        backtest_id,
        history or {},
        results,
        trades,
        results["equity_curve"]
    )

    # 返回文件
    filename = f"backtest_{backtest_id}.xlsx"

    return Response(
        content=excel_data,
        media_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        headers={
            "Content-Disposition": f'attachment; filename="{filename}"'
        }
    )
```

## 三、前端方案（页面模块）

### 3.1 图表导出

**使用ECharts导出PNG**：
```typescript
import * as echarts from 'echarts'

function exportChartToPNG(chartId: string, filename: string): void {
  const chart = echarts.getInstanceByDom(document.getElementById(chartId))

  if (!chart) {
    ElMessage.error('图表不存在')
    return
  }

  // 导出为PNG
  const url = chart.getDataURL({
    type: 'png',
    pixelRatio: 2,  // 2倍分辨率
    backgroundColor: '#fff'
  })

  // 创建下载链接
  const link = document.createElement('a')
  link.href = url
  link.download = `${filename}.png`
  link.click()
}
```

### 3.2 Excel导出按钮

```vue
<template>
  <el-button
    type="primary"
    :loading="exporting"
    @click="handleExportExcel"
  >
    <el-icon><Download /></el-icon>
    导出Excel
  </el-button>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import axios from 'axios'

const props = defineProps<{
  backtestId: string
}>()

const exporting = ref(false)

async function handleExportExcel() {
  exporting.value = true

  try {
    const response = await axios.get(
      `/api/backtest/${props.backtestId}/export/excel`,
      {
        responseType: 'blob'
      }
    )

    // 创建下载链接
    const url = window.URL.createObjectURL(new Blob([response.data]))
    const link = document.createElement('a')
    link.href = url
    link.setAttribute('download', `backtest_${props.backtestId}.xlsx`)
    document.body.appendChild(link)
    link.click()

    // 清理
    document.body.removeChild(link)
    window.URL.revokeObjectURL(url)

    ElMessage.success('导出成功')
  } catch (error) {
    ElMessage.error('导出失败')
    console.error(error)
  } finally {
    exporting.value = false
  }
}
</script>
```
