# 04-回测执行引擎服务研发方案

## 一、领域服务概述

### 1.1 服务定位

**回测执行引擎服务**是股票回测系统的核心服务，负责执行回测任务、模拟交易过程、管理资金持仓。该服务基于策略信号和行情数据，模拟真实交易过程，生成交易记录和资金曲线。

### 1.2 业务边界

**本服务负责**：
- 回测任务的创建和初始化
- 交易模拟执行（基于策略信号）
- 资金持仓管理（现金、持仓股数、持仓成本）
- T+1交易规则实现
- 100股倍数规则约束
- 交易费用计算（佣金、印花税、滑点）
- 任务状态管理（运行中、已完成、已中断）
- 回测任务的暂停和恢复

**本服务不负责**：
- 策略代码执行（由策略管理服务提供策略实例）
- 行情数据获取（由股票数据服务提供）
- 结果计算和展示（由结果计算服务负责）
- 进度实时推送（由进度管理服务负责）

### 1.3 核心能力

1. **回测任务管理**
   - 创建回测任务
   - 初始化回测状态
   - 启动回测执行
   - 暂停回测执行
   - 恢复回测执行
   - 放弃回测任务

2. **交易模拟**
   - 基于策略信号生成交易订单
   - 执行买入交易（检查资金充足性）
   - 执行卖出交易（检查持仓充足性）
   - 记录交易明细

3. **资金持仓管理**
   - 跟踪可用现金
   - 跟踪持仓股数
   - 跟踪持仓成本
   - 计算持仓盈亏

4. **交易规则实现**
   - T+1规则（当日买入的股票不能当日卖出）
   - 100股倍数规则（A股交易必须是100股的整数倍）
   - 涨跌停限制（价格不能超过涨跌停板）

5. **交易费用计算**
   - 佣金计算（万分之2.5，最低5元）
   - 印花税计算（卖出时千分之一）
   - 滑点模拟（买卖价差）

### 1.4 依赖关系

**被依赖方**：
- 进度管理服务（需要回测进度信息）
- 结果计算服务（需要回测结果数据）
- 历史记录管理服务（需要回测结果）

**依赖方**：
- 策略管理服务（获取策略实例）
- 股票数据服务（获取行情数据）
- 交易日历服务（获取交易日列表）

**外部依赖**：
- MongoDB数据库（存储回测任务和交易记录）
- Redis（任务队列和状态缓存）
- Celery（异步任务执行，可选）

import { ref, onMounted, onUnmounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { useBacktestStore } from '@/stores/backtest'
import { useWebSocket } from '@/composables/useWebSocket'

const props = defineProps<Props>()
const emit = defineEmits<Emits>()

const backtestStore = useBacktestStore()
const starting = ref(false)
const controlState = ref<BacktestControlState | null>(null)

// WebSocket连接
const { connect, disconnect, onMessage } = useWebSocket()

// 开始回测
async function handleStart() {
  starting.value = true

  try {
    // 创建回测任务
    const backtestId = await backtestStore.startBacktest(props.parameters)

    // 连接WebSocket获取进度
    connect(`/ws/backtest/${backtestId}/progress`)

    // 监听进度消息
    onMessage((message) => {
      if (message.type === 'progress') {
        controlState.value = message.data
      } else if (message.type === 'complete') {
        emit('complete', message.data.result)
      } else if (message.type === 'error') {
        emit('error', message.data.error)
      }
    })

    emit('start', backtestId)
  } catch (error) {
    ElMessage.error('启动回测失败')
    console.error(error)
  } finally {
    starting.value = false
  }
}

// 暂停回测
async function handlePause() {
  try {
    await backtestStore.pauseBacktest(props.backtestId!)
    emit('pause')
  } catch (error) {
    ElMessage.error('暂停回测失败')
  }
}

// 继续回测
async function handleContinue() {
  try {
    await backtestStore.continueBacktest(props.backtestId!)
    emit('continue')
  } catch (error) {
    ElMessage.error('继续回测失败')
  }
}

// 放弃回测
async function handleAbort() {
  try {
    await ElMessageBox.confirm(
      '确定要放弃当前回测吗？已执行的结果将会丢失。',
      '确认',
      {
        confirmButtonText: '确定',
        cancelButtonText: '取消',
        type: 'warning'
      }
    )

    await backtestStore.abortBacktest(props.backtestId!)
    disconnect()
    emit('abort')
  } catch (error) {
    if (error !== 'cancel') {
      ElMessage.error('放弃回测失败')
    }
  }
}

function progressFormat(percentage: number): string {
  return `${percentage}%`
}

function getStatusType(status: string): string {
  const typeMap: Record<string, string> = {
    'idle': 'info',
    'running': 'primary',
    'paused': 'warning',
    'completed': 'success',
    'error': 'danger'
  }
  return typeMap[status] || 'info'
}

function getStatusText(status: string): string {
  const textMap: Record<string, string> = {
    'idle': '空闲',
    'running': '运行中',
    'paused': '已暂停',
    'completed': '已完成',
    'error': '错误'
  }
  return textMap[status] || '未知'
}

function formatTime(seconds: number): string {
  const hours = Math.floor(seconds / 3600)
  const minutes = Math.floor((seconds % 3600) / 60)
  const secs = seconds % 60

  if (hours > 0) {
    return `${hours}:${String(minutes).padStart(2, '0')}:${String(secs).padStart(2, '0')}`
  }
  return `${minutes}:${String(secs).padStart(2, '0')}`
}

### 2.3 状态管理

#### Pinia Store：backtestStore.ts

```typescript
import { defineStore } from 'pinia'
import { ref } from 'vue'
import axios from 'axios'

export const useBacktestStore = defineStore('backtest', () => {
  // 状态
  const currentBacktestId = ref<string | null>(null)
  const backtestStatus = ref<string>('idle')

  // Actions
  async function startBacktest(parameters: BacktestParameters): Promise<string> {
    try {
      const response = await axios.post('/api/backtest/start', {
        parameters
      })

      currentBacktestId.value = response.data.backtest_id
      backtestStatus.value = 'running'

      return response.data.backtest_id
    } catch (error) {
      console.error('启动回测失败:', error)
      throw error
    }
  }

  async function pauseBacktest(backtestId: string) {
    try {
      await axios.post(`/api/backtest/${backtestId}/interrupt`)
      backtestStatus.value = 'paused'
    } catch (error) {
      console.error('暂停回测失败:', error)
      throw error
    }
  }

  async function continueBacktest(backtestId: string) {
    try {
      await axios.post(`/api/backtest/${backtestId}/continue`)
      backtestStatus.value = 'running'
    } catch (error) {
      console.error('继续回测失败:', error)
      throw error
    }
  }

  async function abortBacktest(backtestId: string) {
    try {
      await axios.delete(`/api/backtest/${backtestId}`)
      currentBacktestId.value = null
      backtestStatus.value = 'idle'
    } catch (error) {
      console.error('放弃回测失败:', error)
      throw error
    }
  }

  async function getBacktestStatus(backtestId: string): Promise<BacktestStatus> {
    try {
      const response = await axios.get(`/api/backtest/${backtestId}/status`)
      backtestStatus.value = response.data.status
      return response.data
    } catch (error) {
      console.error('获取回测状态失败:', error)
      throw error
    }
  }

  return {
    // 状态
    currentBacktestId,
    backtestStatus,
    // Actions
    startBacktest,
    pauseBacktest,
    continueBacktest,
    abortBacktest,
    getBacktestStatus
  }
})
```

### 2.4 WebSocket连接管理

#### Composable：useWebSocket.ts

```typescript
import { onUnmounted } from 'vue'
import { useWebSocket as useWebSocketCore } from '@vueuse/core'

export function useWebSocket() {
  let ws: WebSocket | null = null
  const messageHandlers = new Set<(message: any) => void>()

  function connect(url: string) {
    // 构建完整WebSocket URL
    const wsUrl = `ws://localhost:8000${url}`

    ws = new WebSocket(wsUrl)

    ws.onopen = () => {
      console.log('WebSocket connected:', url)
    }

    ws.onmessage = (event) => {
      try {
        const message = JSON.parse(event.data)
        messageHandlers.forEach(handler => handler(message))
      } catch (error) {
        console.error('Failed to parse WebSocket message:', error)
      }
    }

    ws.onerror = (error) => {
      console.error('WebSocket error:', error)
    }

    ws.onclose = () => {
      console.log('WebSocket disconnected:', url)
    }
  }

  function disconnect() {
    if (ws) {
      ws.close()
      ws = null
    }
    messageHandlers.clear()
  }

  function onMessage(handler: (message: any) => void) {
    messageHandlers.add(handler)

    // 返回清理函数
    return () => {
      messageHandlers.delete(handler)
    }
  }

  function send(message: any) {
    if (ws && ws.readyState === WebSocket.OPEN) {
      ws.send(JSON.stringify(message))
    }
  }

  onUnmounted(() => {
    disconnect()
  })

  return {
    connect,
    disconnect,
    onMessage,
    send
  }
}
```

const progressUpdateTimer = ref<number | null>(null)

onMessage((message) => {

from motor.motor_asyncio import AsyncIOMotorDatabase
from typing import Dict, Any, List
from datetime import datetime, timedelta
import asyncio

class BacktestEngine:
    """回测执行引擎"""

    def __init__(self, db: AsyncIOMotorDatabase):
        self.db = db
        self.should_pause = False
        self.should_abort = False

    async def execute_backtest(
        self,
        backtest_id: str,
        parameters: Dict[str, Any]
    ):
        """
        执行回测任务

        Args:
            backtest_id: 回测任务ID
            parameters: 回测参数
        """
        try:
            # 1. 更新任务状态为运行中
            await self.db.backtest_tasks.update_one(
                {"backtest_id": backtest_id},
                {"$set": {"status": "running"}}
            )

            # 2. 获取行情数据
            quotes = await self._get_quotes(parameters)
            trading_days = await self._get_trading_days(parameters)

            # 3. 初始化回测状态
            state = BacktestState(
                initial_capital=parameters["initial_capital"],
                quotes=quotes,
                trading_days=trading_days
            )

            # 4. 创建策略实例
            strategy = self._create_strategy(parameters["strategy_id"], parameters["strategy_params"])

            # 5. 执行回测循环
            for i, trading_day in enumerate(trading_days):
                # 检查是否需要暂停
                if self.should_pause:
                    await self._pause_backtest(backtest_id, state, i)
                    break

                # 检查是否需要放弃
                if self.should_abort:
                    await self._abort_backtest(backtest_id)
                    return

                # 获取当日行情
                quote = self._get_quote_by_date(quotes, trading_day)
                if not quote:
                    continue

                # 执行策略
                signal = strategy.on_bar(
                    bar_id=f"{backtest_id}_{i}",
                    timestamp=datetime.strptime(trading_day, "%Y-%m-%d"),
                    current_price=quote["close"],
                    position=state.position,
                    cash=state.cash
                )

                # 执行交易
                if signal["action"] in ["buy", "sell"]:
                    await self._execute_trade(backtest_id, state, signal, quote, trading_day)

                # 更新每日状态
                await self._update_daily_state(backtest_id, state, quote, trading_day, i)

                # 更新进度
                progress = (i + 1) / len(trading_days) * 100
                await self._update_progress(backtest_id, i, len(trading_days), trading_day, progress)

            # 6. 完成回测
            if not self.should_pause and not self.should_abort:
                await self._complete_backtest(backtest_id, state)

        except Exception as e:
            await self._handle_error(backtest_id, e)

    async def _get_quotes(self, parameters: Dict[str, Any]) -> List[Dict]:
        """获取行情数据"""
        from services.stock_data_service import StockDataService
        stock_data_service = StockDataService(self.db, None)

        return await stock_data_service.get_quotes(
            parameters["stock_code"],
            parameters["start_date"],
            parameters["end_date"]
        )

    async def _get_trading_days(self, parameters: Dict[str, Any]) -> List[str]:
        """获取交易日列表"""
        from services.trading_calendar_service import TradingCalendarService
        trading_calendar_service = TradingCalendarService(self.db, None)

        return await trading_calendar_service.get_trading_days(
            parameters["start_date"],
            parameters["end_date"]
        )

    def _create_strategy(self, strategy_id: str, strategy_params: Dict[str, Any]):
        """创建策略实例"""
        from core.strategies.registry import StrategyRegistry
        strategy_class = StrategyRegistry.get_strategy(strategy_id)
        return strategy_class(strategy_params)

    def _get_quote_by_date(self, quotes: List[Dict], date: str) -> Dict:
        """根据日期获取行情"""
        for quote in quotes:
            if quote["date"] == date:
                return quote
        return None

    async def _execute_trade(
        self,
        backtest_id: str,
        state: 'BacktestState',
        signal: Dict,
        quote: Dict,
        date: str
    ):
        """执行交易"""
        if signal["action"] == "buy":
            await self._execute_buy(state, signal, quote, date)
        elif signal["action"] == "sell":
            await self._execute_sell(state, signal, quote, date)

    async def _execute_buy(
        self,
        state: 'BacktestState',
        signal: Dict,
        quote: Dict,
        date: str
    ):
        """执行买入"""
        price = quote["close"]
        max_shares = int(state.cash / price / 100) * 100  # 100股倍数

        if max_shares == 0:
            return

        shares = min(signal.get("amount", max_shares), max_shares)
        amount = shares * price

        # 计算费用
        commission = max(amount * 0.00025, 5)  # 佣金
        total_cost = amount + commission

        if total_cost > state.cash:
            return  # 资金不足

        # 更新状态
        state.cash -= total_cost
        state.position += shares
        state.position_cost = (state.position_cost * (state.position - shares) + price * shares) / state.position

        # 记录交易
        trade_doc = {
            "backtest_id": state.backtest_id,
            "date": date,
            "trade_type": "buy",
            "price": price,
            "shares": shares,
            "amount": amount,
            "commission": commission,
            "stamp_duty": 0,
            "slippage": 0,
            "total_cost": total_cost,
            "cash_before": state.cash + total_cost,
            "cash_after": state.cash,
            "position_before": state.position - shares,
            "position_after": state.position,
            "signal": signal,
            "created_at": datetime.utcnow()
        }

        await self.db.backtest_trades.insert_one(trade_doc)

    async def _execute_sell(
        self,
        state: 'BacktestState',
        signal: Dict,
        quote: Dict,
        date: str
    ):
        """执行卖出"""
        if state.position == 0:
            return  # 无持仓

        price = quote["close"]
        shares = min(signal.get("amount", state.position), state.position)

        # T+1检查：当日买入的不能卖出
        # 简化实现：这里假设所有持仓都可以卖出
        # 实际需要跟踪每笔买入的日期

        shares = (shares // 100) * 100  # 100股倍数

        if shares == 0:
            return

        amount = shares * price

        # 计算费用
        commission = max(amount * 0.00025, 5)
        stamp_duty = amount * 0.001  # 印花税
        total_cost = commission + stamp_duty

        # 更新状态
        state.cash += amount - total_cost
        state.position -= shares

        # 记录交易
        trade_doc = {
            "backtest_id": state.backtest_id,
            "date": date,
            "trade_type": "sell",
            "price": price,
            "shares": shares,
            "amount": amount,
            "commission": commission,
            "stamp_duty": stamp_duty,
            "slippage": 0,
            "total_cost": total_cost,
            "cash_before": state.cash - amount + total_cost,
            "cash_after": state.cash,
            "position_before": state.position + shares,
            "position_after": state.position,
            "signal": signal,
            "created_at": datetime.utcnow()
        }

        await self.db.backtest_trades.insert_one(trade_doc)

    async def _update_daily_state(
        self,
        backtest_id: str,
        state: 'BacktestState',
        quote: Dict,
        date: str,
        bar_index: int
    ):

        market_value = state.position * quote["close"]
        total_assets = state.cash + market_value
        profit_loss = market_value - state.position * state.position_cost if state.position > 0 else 0

        daily_state_doc = {
            "backtest_id": backtest_id,
            "date": date,
            "bar_index": bar_index,
            "cash": state.cash,
            "position": state.position,
            "position_cost": state.position_cost,
            "current_price": quote["close"],
            "market_value": market_value,
            "total_assets": total_assets,
            "profit_loss": profit_loss,
            "created_at": datetime.utcnow()
        }

        await self.db.backtest_daily_states.insert_one(daily_state_doc)

    async def _update_progress(

#### 3.4.1 交易规则实现

**T+1规则**：
```python
class BacktestState:
    def __init__(self, ...):
        # ...
        self.position_lots = []  # 持仓批次列表

    def add_position(self, shares: int, cost: float, date: str):
        """添加持仓"""
        self.position_lots.append({
            "shares": shares,
            "cost": cost,
            "buy_date": date  # 买入日期
        })
        self.position += shares

    def can_sell(self, shares: int, current_date: str) -> bool:
        """检查是否可以卖出（T+1规则）"""
        sellable_shares = 0
        for lot in self.position_lots:
            # 检查是否可以卖出（买入日期不能是当前日期）
            if lot["buy_date"] != current_date:
                sellable_shares += lot["shares"]

        return sellable_shares >= shares

    def sell_position(self, shares: int) -> float:
        """卖出持仓（返回持仓成本）"""
        remaining_shares = shares
        cost_basis = 0

        for lot in self.position_lots[:]:
            if remaining_shares == 0:
                break

            if lot["shares"] <= remaining_shares:
                # 完全卖出这个批次
                cost_basis += lot["shares"] * lot["cost"]
                remaining_shares -= lot["shares"]
                self.position_lots.remove(lot)
            else:
                # 部分卖出这个批次
                cost_basis += remaining_shares * lot["cost"]
                lot["shares"] -= remaining_shares
                remaining_shares = 0

        return cost_basis
```

```python
class TradingCostCalculator:
    """交易费用计算器"""

    @staticmethod
    def calculate_buy_cost(amount: float) -> dict:
        """计算买入费用"""
        # 佣金：万分之2.5，最低5元
        commission = max(amount * 0.00025, 5)

        # 过户费：万分之0.2（深圳）
        transfer_fee = amount * 0.00002

        total_cost = commission + transfer_fee

        return {
            "commission": commission,
            "transfer_fee": transfer_fee,
            "stamp_duty": 0,  # 买入无印花税
            "total_cost": total_cost
        }

    @staticmethod
    def calculate_sell_cost(amount: float) -> dict:
        """计算卖出费用"""
        # 佣金：万分之2.5，最低5元
        commission = max(amount * 0.00025, 5)

        # 印花税：千分之一（仅卖出）
        stamp_duty = amount * 0.001

        # 过户费：万分之0.2
        transfer_fee = amount * 0.00002

        total_cost = commission + stamp_duty + transfer_fee

        return {
            "commission": commission,
            "stamp_duty": stamp_duty,
            "transfer_fee": transfer_fee,
            "total_cost": total_cost
        }
```

from fastapi import WebSocket
from typing import Set

class ConnectionManager:
    """WebSocket连接管理器"""

    def __init__(self):
        self.active_connections: dict[str, list[WebSocket]] = {}

    async def connect(self, websocket: WebSocket, backtest_id: str):
        await websocket.accept()
        if backtest_id not in self.active_connections:
            self.active_connections[backtest_id] = []
        self.active_connections[backtest_id].append(websocket)

    def disconnect(self, websocket: WebSocket, backtest_id: str):
        self.active_connections[backtest_id].remove(websocket)

    async def send_progress(self, backtest_id: str, progress_data: dict):
        """发送进度更新"""
        if backtest_id in self.active_connections:
            message = {
                "type": "progress",
                "data": progress_data
            }

            # 广播给所有连接的客户端
            for connection in self.active_connections[backtest_id]:
                await connection.send_json(message)

    async def send_position(self, backtest_id: str, position_data: dict):
        """发送持仓更新"""
        if backtest_id in self.active_connections:
            message = {
                "type": "position_update",
                "data": position_data
            }

            for connection in self.active_connections[backtest_id]:
                await connection.send_json(message)


manager = ConnectionManager()

    try:
        while True:
            # 保持连接
            await websocket.receive_text()
