# 05-进度管理服务研发方案

## 一、领域服务概述

### 1.1 服务定位

**进度管理服务**负责回测进度的实时跟踪和推送，通过WebSocket向前端推送回测进度、持仓状态等信息，确保用户能够实时了解回测执行情况。

### 1.2 业务边界

**本服务负责**：
- WebSocket连接管理
- 实时进度推送（当前K线、进度百分比、时间信息）
- 持仓状态推送（现金、持仓、盈亏）
- 交易信号推送（买卖信号）
- 错误信息推送
- HTTP轮询备用接口

**本服务不负责**：
- 回测任务执行（由回测执行引擎负责）
- 进度数据存储（进度数据存储在backtest_tasks中）

### 1.3 核心能力

1. **WebSocket连接管理**
   - 连接建立与认证
   - 心跳保活
   - 连接断开与重连
   - 多客户端支持

2. **进度推送**
   - 回测进度实时更新
   - 持仓状态实时更新
   - 交易信号实时推送
   - 错误信息实时推送

3. **HTTP轮询支持**
   - 提供HTTP接口获取当前状态
   - 作为WebSocket的备用方案

### 1.4 依赖关系

**被依赖方**：
- 前端回测页面（需要实时进度）

**依赖方**：
- 回测执行引擎服务（需要推送进度）

**外部依赖**：
- Redis（WebSocket连接状态管理）

```typescript
import { ref, onMounted, onUnmounted } from 'vue'

const RECONNECT_INTERVAL = 3000 // 3秒重连间隔
const HEARTBEAT_INTERVAL = 30000 // 30秒心跳

export function useProgressWebSocket(backtestId: string) {
  const ws = ref<WebSocket | null>(null)
  const reconnectTimer = ref<number | null>(null)
  const heartbeatTimer = ref<number | null>(null)
  const messageHandlers = new Map<string, (data: any) => void>()

  function connect() {
    const wsUrl = `ws://localhost:8000/ws/backtest/${backtestId}/progress`
    ws.value = new WebSocket(wsUrl)

    ws.value.onopen = () => {
      console.log('WebSocket connected')
      startHeartbeat()
    }

    ws.value.onmessage = (event) => {
      try {
        const message = JSON.parse(event.data)
        const handler = messageHandlers.get(message.type)
        if (handler) {
          handler(message.data)
        }
      } catch (error) {
        console.error('Failed to parse message:', error)
      }
    }

    ws.value.onerror = (error) => {
      console.error('WebSocket error:', error)
    }

    ws.value.onclose = () => {
      console.log('WebSocket closed, reconnecting...')
      stopHeartbeat()
      scheduleReconnect()
    }
  }

  function disconnect() {
    if (ws.value) {
      ws.value.close()
      ws.value = null
    }
    stopHeartbeat()
    if (reconnectTimer.value) {
      clearTimeout(reconnectTimer.value)
      reconnectTimer.value = null
    }
  }

  function scheduleReconnect() {
    reconnectTimer.value = window.setTimeout(() => {
      connect()
    }, RECONNECT_INTERVAL)
  }

  function startHeartbeat() {
    heartbeatTimer.value = window.setInterval(() => {
      send({ type: 'ping' })
    }, HEARTBEAT_INTERVAL)
  }

  function stopHeartbeat() {
    if (heartbeatTimer.value) {
      clearInterval(heartbeatTimer.value)
      heartbeatTimer.value = null
    }
  }

  function send(message: any) {
    if (ws.value && ws.value.readyState === WebSocket.OPEN) {
      ws.value.send(JSON.stringify(message))
    }
  }

  function onMessageType(type: string, handler: (data: any) => void) {
    messageHandlers.set(type, handler)
  }

  onUnmounted(() => {
    disconnect()
  })

  return {
    connect,
    disconnect,
    send,
    onMessageType
  }
}
```

## 三、后端方案（领域服务）

### 3.1 WebSocket连接管理器

**实现**：
```python
from fastapi import WebSocket, WebSocketDisconnect
from typing import Dict, List
import json
import asyncio

class ConnectionManager:
    """WebSocket连接管理器"""

    def __init__(self):
        # backtest_id -> list of WebSocket connections
        self.active_connections: Dict[str, List[WebSocket]] = {}
        # WebSocket -> backtest_id (用于反向查找)
        self.connection_backtest_map: Dict[WebSocket, str] = {}

    async def connect(self, websocket: WebSocket, backtest_id: str):
        """接受连接"""
        await websocket.accept()

        if backtest_id not in self.active_connections:
            self.active_connections[backtest_id] = []

        self.active_connections[backtest_id].append(websocket)
        self.connection_backtest_map[websocket] = backtest_id

        # 发送连接确认消息
        await websocket.send_json({
            "type": "connected",
            "data": {
                "backtest_id": backtest_id,
                "timestamp": datetime.utcnow().isoformat()
            }
        })

    def disconnect(self, websocket: WebSocket):
        """移除断开的连接"""
        if websocket in self.connection_backtest_map:
            backtest_id = self.connection_backtest_map[websocket]

            if backtest_id in self.active_connections:
                self.active_connections[backtest_id].remove(websocket)

            del self.connection_backtest_map[websocket]

    async def send_personal_message(self, message: dict, websocket: WebSocket):
        """发送消息给特定客户端"""
        try:
            await websocket.send_json(message)
        except Exception as e:
            print(f"Failed to send message: {e}")
            self.disconnect(websocket)

    async def broadcast_to_backtest(self, backtest_id: str, message: dict):
        """广播消息给回测任务的所有订阅者"""
        if backtest_id in self.active_connections:
            # 创建发送任务
            tasks = []
            for connection in self.active_connections[backtest_id]:
                tasks.append(self.send_personal_message(message, connection))

            # 并发发送
            if tasks:
                await asyncio.gather(*tasks, return_exceptions=True)

    async def broadcast_progress(
        self,
        backtest_id: str,
        current_bar: int,
        total_bars: int,
        current_date: str,
        elapsed_time: float
    ):
        """广播进度更新"""
        progress = (current_bar / total_bars) * 100 if total_bars > 0 else 0
        estimated_time_remaining = None

        if elapsed_time > 0 and current_bar > 0:
            avg_time_per_bar = elapsed_time / current_bar
            remaining_bars = total_bars - current_bar
            estimated_time_remaining = avg_time_per_bar * remaining_bars

        message = {
            "type": "progress",
            "data": {
                "backtest_id": backtest_id,
                "progress": {
                    "current_bar": current_bar,
                    "total_bars": total_bars,
                    "percentage": round(progress, 2),
                    "current_date": current_date,
                    "elapsed_time": elapsed_time,
                    "estimated_time_remaining": round(estimated_time_remaining, 2) if estimated_time_remaining else None
                }
            }
        }

        await self.broadcast_to_backtest(backtest_id, message)

    async def broadcast_position(
        self,
        backtest_id: str,
        cash: float,
        position: int,
        current_price: float,
        total_assets: float
    ):
        """广播持仓更新"""
        market_value = position * current_price

        message = {
            "type": "position_update",
            "data": {
                "backtest_id": backtest_id,
                "position": {
                    "cash": round(cash, 2),
                    "position": position,
                    "current_price": round(current_price, 2),
                    "market_value": round(market_value, 2),
                    "total_assets": round(total_assets, 2)
                }
            }
        }

        await self.broadcast_to_backtest(backtest_id, message)

    async def broadcast_trade(
        self,
        backtest_id: str,
        trade_type: str,
        price: float,
        shares: int,
        amount: float
    ):
        """广播交易信号"""
        message = {
            "type": "trade_signal",
            "data": {
                "backtest_id": backtest_id,
                "trade": {
                    "type": trade_type,
                    "price": round(price, 2),
                    "shares": shares,
                    "amount": round(amount, 2)
                }
            }
        }

        await self.broadcast_to_backtest(backtest_id, message)

    async def broadcast_complete(
        self,
        backtest_id: str,
        result: dict
    ):
        """广播完成消息"""
        message = {
            "type": "complete",
            "data": {
                "backtest_id": backtest_id,
                "result": result
            }
        }

        await self.broadcast_to_backtest(backtest_id, message)

    async def broadcast_error(
        self,
        backtest_id: str,
        error: dict
    ):
        """广播错误消息"""
        message = {
            "type": "error",
            "data": {
                "backtest_id": backtest_id,
                "error": error
            }
        }

        await self.broadcast_to_backtest(backtest_id, message)
```

### 3.2 WebSocket路由

**实现**：
```python
from fastapi import APIRouter, WebSocket, WebSocketDisconnect, Depends
from motor.motor_asyncio import AsyncIOMotorDatabase

router = APIRouter(tags=["websocket"])

@router.websocket("/ws/backtest/{backtest_id}/progress")
async def websocket_progress(
    websocket: WebSocket,
    backtest_id: str,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """
    WebSocket进度推送端点

    推送回测进度、持仓状态等信息
    """
    # 验证回测任务是否存在
    task = await db.backtest_tasks.find_one({"backtest_id": backtest_id})
    if not task:
        await websocket.close(code=1008, reason="Backtest task not found")
        return

    # 接受连接
    await manager.connect(websocket, backtest_id)

    try:
        while True:
            # 接收客户端消息（用于心跳保活）
            data = await websocket.receive_text()

            # 处理心跳
            try:
                message = json.loads(data)
                if message.get("type") == "ping":
                    await websocket.send_json({"type": "pong"})
            except json.JSONDecodeError:
                pass

    except WebSocketDisconnect:
        manager.disconnect(websocket)
    except Exception as e:
        print(f"WebSocket error: {e}")
        manager.disconnect(websocket)
```

### 3.3 HTTP轮询备用接口

**实现**：

```python
@router.get("/backtest/{backtest_id}/current-state")
async def get_current_state(
    backtest_id: str,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """
    获取回测当前状态（HTTP轮询备用）

    当WebSocket不可用时，前端可以使用此接口轮询状态
    """
    task = await db.backtest_tasks.find_one({"backtest_id": backtest_id})

    if not task:
        raise HTTPException(status_code=404, detail="Backtest task not found")

    # 获取最新状态
    execution_info = task.get("execution_info", {})

    result = {
        "backtest_id": backtest_id,
        "status": task["status"],
        "progress": execution_info
    }

    return result
```
