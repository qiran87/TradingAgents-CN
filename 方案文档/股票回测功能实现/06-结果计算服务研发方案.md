# 06-结果计算服务研发方案

## 一、领域服务概述

### 1.1 服务定位

**结果计算服务**负责回测完成后的结果计算和统计分析，生成收益率、最大回撤、胜率等关键指标，以及交易明细和资金曲线数据。

### 1.2 业务边界

**本服务负责**：
- 回测结果指标计算（收益率、最大回撤、夏普比率等）
- 交易明细汇总
- 资金曲线计算
- 结果数据持久化
- 结果数据查询

**本服务不负责**：
- 回测执行（由回测执行引擎负责）
- 结果可视化（由前端负责）

### 1.3 核心能力

1. **收益指标计算**
   - 总收益率
   - 年化收益率
   - 累计收益率曲线
   - 日收益率分布

2. **风险指标计算**
   - 最大回撤
   - 波动率
   - 下行波动率
   - VaR（风险价值）

3. **风险调整收益指标**
   - 夏普比率
   - 索提诺比率
   - 卡玛比率

4. **交易统计**
   - 总交易次数
   - 胜率
   - 盈亏比
   - 平均盈利/亏损

5. **资金曲线**
   - 每日资产变化
   - 持仓市值变化
   - 现金变化

### 2.2 核心服务实现

#### 服务类：services/result_calculator.py

```python
import numpy as np
from typing import Dict, List, Any

class ResultCalculator:
    """回测结果计算器"""

    def calculate_all_metrics(
        self,
        backtest_id: str,
        daily_states: List[Dict],
        trades: List[Dict]
    ) -> Dict[str, Any]:
        """计算所有指标"""
        return {
            "return_metrics": self._calculate_return_metrics(daily_states),
            "risk_metrics": self._calculate_risk_metrics(daily_states),
            "risk_adjusted_metrics": self._calculate_risk_adjusted_metrics(daily_states),
            "trading_stats": self._calculate_trading_stats(trades),
            "equity_curve": self._calculate_equity_curve(daily_states)
        }

    def _calculate_return_metrics(self, daily_states: List[Dict]) -> Dict:
        """计算收益指标"""
        total_assets = [state["total_assets"] for state in daily_states]
        initial_capital = total_assets[0] if total_assets else 100000

        # 总收益率
        final_assets = total_assets[-1] if total_assets else initial_capital
        total_return = (final_assets - initial_capital) / initial_capital

        # 日收益率
        daily_returns = []
        for i in range(1, len(total_assets)):
            daily_return = (total_assets[i] - total_assets[i-1]) / total_assets[i-1]
            daily_returns.append(daily_return)

        # 累计收益率
        cumulative_returns = []
        cumulative_return = 0
        for daily_return in daily_returns:
            cumulative_return = (1 + cumulative_return) * (1 + daily_return) - 1
            cumulative_returns.append(cumulative_return)

        # 年化收益率
        days = len(daily_states)
        annual_return = (1 + total_return) ** (365 / days) - 1 if days > 0 else 0

        return {
            "total_return": round(total_return, 4),
            "annual_return": round(annual_return, 4),
            "cumulative_returns": cumulative_returns,
            "daily_returns": daily_returns
        }

    def _calculate_risk_metrics(self, daily_states: List[Dict]) -> Dict:
        """计算风险指标"""
        total_assets = [state["total_assets"] for state in daily_states]
        initial_capital = total_assets[0] if total_assets else 100000

        # 计算最大回撤
        max_drawdown = 0
        peak = initial_capital

        for assets in total_assets:
            if assets > peak:
                peak = assets

            drawdown = (peak - assets) / peak
            if drawdown > max_drawdown:
                max_drawdown = drawdown

        # 日收益率
        daily_returns = []
        for i in range(1, len(total_assets)):
            daily_return = (total_assets[i] - total_assets[i-1]) / total_assets[i-1]
            daily_returns.append(daily_return)

        # 波动率（年化）
        if daily_returns:
            volatility = np.std(daily_returns) * np.sqrt(252)
        else:
            volatility = 0

        # 下行波动率
        negative_returns = [r for r in daily_returns if r < 0]
        if negative_returns:
            downside_volatility = np.std(negative_returns) * np.sqrt(252)
        else:
            downside_volatility = 0

        # VaR 95%
        if daily_returns:
            var_95 = np.percentile(daily_returns, 5)
        else:
            var_95 = 0

        return {
            "max_drawdown": round(max_drawdown, 4),
            "volatility": round(volatility, 4),
            "downside_volatility": round(downside_volatility, 4),
            "var_95": round(var_95, 4)
        }

    def _calculate_risk_adjusted_metrics(self, daily_states: List[Dict]) -> Dict:
        """计算风险调整收益指标"""
        return_metrics = self._calculate_return_metrics(daily_states)
        risk_metrics = self._calculate_risk_metrics(daily_states)

        # 夏普比率（假设无风险利率为3%）
        risk_free_rate = 0.03
        excess_return = return_metrics["annual_return"] - risk_free_rate
        sharpe_ratio = excess_return / risk_metrics["volatility"] if risk_metrics["volatility"] > 0 else 0

        # 索提诺比率
        sortino_ratio = excess_return / risk_metrics["downside_volatility"] if risk_metrics["downside_volatility"] > 0 else 0

        # 卡玛比率
        calmar_ratio = return_metrics["annual_return"] / risk_metrics["max_drawdown"] if risk_metrics["max_drawdown"] > 0 else 0

        return {
            "sharpe_ratio": round(sharpe_ratio, 4),
            "sortino_ratio": round(sortino_ratio, 4),
            "calmar_ratio": round(calmar_ratio, 4)
        }

    def _calculate_trading_stats(self, trades: List[Dict]) -> Dict:
        """计算交易统计"""
        buy_trades = [t for t in trades if t["trade_type"] == "buy"]
        sell_trades = [t for t in trades if t["trade_type"] == "sell"]

        # 配对买卖交易计算盈亏
        paired_trades = self._pair_trades(buy_trades, sell_trades)

        total_trades = len(paired_trades)
        winning_trades = [t for t in paired_trades if t["profit"] > 0]
        losing_trades = [t for t in paired_trades if t["profit"] < 0]

        win_rate = len(winning_trades) / total_trades if total_trades > 0 else 0

        avg_profit = np.mean([t["profit"] for t in winning_trades]) if winning_trades else 0
        avg_loss = np.mean([t["profit"] for t in losing_trades]) if losing_trades else 0

        profit_loss_ratio = abs(avg_profit / avg_loss) if avg_loss != 0 else 0

        return {
            "total_trades": total_trades,
            "winning_trades": len(winning_trades),
            "losing_trades": len(losing_trades),
            "win_rate": round(win_rate, 4),
            "avg_profit": round(avg_profit, 2),
            "avg_loss": round(avg_loss, 2),
            "profit_loss_ratio": round(profit_loss_ratio, 4)
        }

    def _pair_trades(self, buy_trades: List[Dict], sell_trades: List[Dict]) -> List[Dict]:
        """配对买卖交易"""
        paired = []

        # 简化实现：按顺序配对
        buy_index = 0
        sell_index = 0

        while buy_index < len(buy_trades) and sell_index < len(sell_trades):
            buy_trade = buy_trades[buy_index]
            sell_trade = sell_trades[sell_index]

            # 计算盈亏
            profit = (sell_trade["price"] - buy_trade["price"]) * sell_trade["shares"]
            profit -= (buy_trade["total_cost"] + sell_trade["total_cost"])

            paired.append({
                "buy_date": buy_trade["date"],
                "sell_date": sell_trade["date"],
                "buy_price": buy_trade["price"],
                "sell_price": sell_trade["price"],
                "shares": sell_trade["shares"],
                "profit": profit
            })

            buy_index += 1
            sell_index += 1

        return paired

    def _calculate_equity_curve(self, daily_states: List[Dict]) -> Dict:
        """计算资金曲线"""
        dates = [state["date"] for state in daily_states]
        total_assets = [state["total_assets"] for state in daily_states]
        cash = [state["cash"] for state in daily_states]
        position_value = [state["market_value"] for state in daily_states]

        return {
            "dates": dates,
            "total_assets": total_assets,
            "cash": cash,
            "position_value": position_value
        }
```

### 2.3 API接口

```python
@router.get("/backtest/{backtest_id}/results")
async def get_backtest_results(
    backtest_id: str,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """获取回测结果"""
    result = await db.backtest_results.find_one({"backtest_id": backtest_id})

    if not result:
        raise HTTPException(status_code=404, detail="Results not found")

    return result

@router.get("/backtest/{backtest_id}/trades")
async def get_backtest_trades(
    backtest_id: str,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """获取交易明细"""
    cursor = db.backtest_trades.find({"backtest_id": backtest_id}).sort("date", 1)
    trades = await cursor.to_list(length=1000)

    return {"trades": trades}

@router.get("/backtest/{backtest_id}/equity-curve")
async def get_equity_curve(
    backtest_id: str,
    db: AsyncIOMotorDatabase = Depends(get_database)
):
    """获取资金曲线"""
    result = await db.backtest_results.find_one({"backtest_id": backtest_id})

    if not result:
        raise HTTPException(status_code=404, detail="Results not found")

    return result["equity_curve"]
```
