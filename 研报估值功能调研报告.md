# 研报估值功能调研报告

## 📋 调研目标

调研 TradingAgents-CN 平台是否存储不同券商/机构的研报估值数据(如目标价、评级等),为后续功能开发提供技术参考。

---

## 🔍 调研结论

### 核心发现

**当前平台未存储第三方研报估值数据**

- ✅ 平台具备 **AI 分析师** 模块,可生成估值建议
- ❌ 没有存储券商/机构发布的研报估值数据
- ❌ 没有存储分析师评级、目标价等信息
- ⚠️ Akshare 库提供研报数据接口,但未集成

#### 核心服务类

**文件**: `app/services/research_report_service.py`

```python
from typing import List, Optional
from datetime import datetime, timedelta
from motor.motor_asyncio import AsyncIOMotorDatabase
from app.core.rating_normalizer import RatingNormalizer
import akshare as ak

class ResearchReportService:
    """研报服务"""

    def __init__(self, db: AsyncIOMotorDatabase):
        self.db = db
        self.collection = db.stock_research_reports
        self.rating_normalizer = RatingNormalizer(db)

    async def fetch_reports(
        self,
        stock_code: str,
        days: int = 90
    ) -> List[dict]:
        """
        从 Akshare 获取研报数据

        Args:
            stock_code: 股票代码
            days: 获取最近 N 天的研报

        Returns:
            研报数据列表
        """
        try:
            df = ak.stock_research_report_em(
                symbol=stock_code,
                indicator="按个股"
            )

            # 转换为字典列表
            reports = df.to_dict("records")

            # 标准化评级
            for report in reports:
                report["rating_code"] = await self.rating_normalizer.normalize(
                    institution=report.get("institution", "东方财富"),
                    original_rating=report.get("rating", "")
                )

            return reports

        except Exception as e:
            print(f"获取研报失败: {e}")
            return []

    async def save_reports(self, reports: List[dict]) -> int:
        """
        保存研报到数据库 (去重)

        Args:
            reports: 研报列表

        Returns:
            新增数量
        """
        saved_count = 0

        for report in reports:
            try:
                # 检查是否已存在
                existing = await self.collection.find_one({
                    "stock_code": report["stock_code"],
                    "institution": report["institution"],
                    "report_date": report["research_report_date"]
                })

                if not existing:
                    # 插入新研报
                    await self.collection.insert_one({
                        "stock_code": report["stock_code"],
                        "stock_name": report["stock_name"],
                        "report_date": datetime.strptime(
                            report["research_report_date"],
                            "%Y-%m-%d"
                        ),
                        "institution": report["institution"],
                        "analyst": report.get("analyst", ""),
                        "rating": report.get("rating", ""),
                        "rating_code": report.get("rating_code", ""),
                        "target_price": float(report.get("target_price", 0)) if report.get("target_price") else None,
                        "report_title": report.get("report_title", ""),
                        "report_url": report.get("report_url", ""),
                        "data_source": "eastmoney",
                        "created_at": datetime.utcnow(),
                        "updated_at": datetime.utcnow(),
                        "is_valid": True
                    })
                    saved_count += 1

            except Exception as e:
                print(f"保存研报失败: {e}")
                continue

        return saved_count

    async def get_reports_by_stock(
        self,
        stock_code: str,
        days: int = 90,
        limit: int = 50
    ) -> List[dict]:
        """
        查询指定股票的研报

        Args:
            stock_code: 股票代码
            days: 查询最近 N 天
            limit: 返回条数

        Returns:
            研报列表
        """
        since_date = datetime.utcnow() - timedelta(days=days)

        cursor = self.collection.find({
            "stock_code": stock_code,
            "report_date": {"$gte": since_date},
            "is_valid": True
        }).sort("report_date", -1).limit(limit)

        return await cursor.to_list(length=limit)

    async def get_consensus_rating(
        self,
        stock_code: str,
        days: int = 90
    ) -> dict:
        """
        计算市场共识评级

        Args:
            stock_code: 股票代码
            days: 统计周期

        Returns:
            共识评级数据
        """
        reports = await self.get_reports_by_stock(stock_code, days, limit=1000)

        if not reports:
            return {
                "stock_code": stock_code,
                "total_reports": 0,
                "average_rating": None,
                "consensus": "无数据"
            }

        # 统计评级分布
        rating_counts = {}
        total_rating_score = 0
        valid_target_prices = []

        for report in reports:
            rating = report.get("rating_code", "HOLD")
            rating_counts[rating] = rating_counts.get(rating, 0) + 1

            # 计算平均评级分数
            rating_score = self._get_rating_score(rating)
            total_rating_score += rating_score

            if report.get("target_price"):
                valid_target_prices.append(report["target_price"])

        # 计算共识评级
        avg_rating_score = total_rating_score / len(reports)
        consensus = self._score_to_rating(avg_rating_score)

        # 计算平均目标价
        avg_target_price = (
            sum(valid_target_prices) / len(valid_target_prices)
            if valid_target_prices else None
        )

        return {
            "stock_code": stock_code,
            "total_reports": len(reports),
            "rating_distribution": rating_counts,
            "average_rating_score": round(avg_rating_score, 2),
            "consensus": consensus,
            "average_target_price": round(avg_target_price, 2) if avg_target_price else None,
            "min_target_price": min(valid_target_prices) if valid_target_prices else None,
            "max_target_price": max(valid_target_prices) if valid_target_prices else None,
            "report_count": len(valid_target_prices)
        }

    def _get_rating_score(self, rating_code: str) -> int:
        """将评级代码转换为分数"""
        score_map = {
            "BUY": 5,
            "OVERWEIGHT": 4,
            "HOLD": 3,
            "UNDERWEIGHT": 2,
            "SELL": 1
        }
        return score_map.get(rating_code, 3)

    def _score_to_rating(self, score: float) -> str:
        """将分数转换为评级"""
        if score >= 4.5:
            return "强烈买入"
        elif score >= 3.5:
            return "买入"
        elif score >= 2.5:
            return "持有"
        elif score >= 1.5:
            return "减持"
        else:
            return "卖出"
```

### 3. 定时同步任务

**文件**: `app/worker/research_report_sync.py`

```python
from apscheduler.schedulers.asyncio import AsyncIOScheduler
from app.services.research_report_service import ResearchReportService
from app.core.database import get_database

class ResearchReportSync:
    """研报同步任务"""

    def __init__(self):
        self.scheduler = AsyncIOScheduler()
        self.service = None

    async def sync_all_stocks(self):
        """同步所有股票的研报"""
        db = get_database()
        self.service = ResearchReportService(db)

        # 从 stock_basic_info 获取股票列表
        cursor = db.stock_basic_info.find(
            {"list_status": "L"},  # 仅限正常上市股票
            {"symbol": 1}
        )

        stocks = await cursor.to_list(length=5000)
        total_saved = 0

        for stock in stocks:
            stock_code = stock["symbol"]

            # 获取并保存研报
            reports = await self.service.fetch_reports(stock_code)
            saved = await self.service.save_reports(reports)
            total_saved += saved

            print(f"✅ {stock_code}: 获取 {len(reports)} 条研报, 新增 {saved} 条")

        print(f"\n🎉 同步完成! 总共新增 {total_saved} 条研报")

    async def sync_single_stock(self, stock_code: str):
        """同步单只股票的研报"""
        db = get_database()
        self.service = ResearchReportService(db)

        reports = await self.service.fetch_reports(stock_code)
        saved = await self.service.save_reports(reports)

        print(f"✅ {stock_code}: 获取 {len(reports)} 条研报, 新增 {saved} 条")

        return saved

    def start(self):
        """启动定时任务"""

        # 每周日凌晨 2 点全量同步
        self.scheduler.add_job(
            self.sync_all_stocks,
            "cron",
            day_of_week="sun",
            hour=2,
            minute=0,
            id="sync_all_reports"
        )

        # 每天凌晨 1 点同步热门股票
        self.scheduler.add_job(
            self.sync_hot_stocks,
            "cron",
            hour=1,
            minute=0,
            id="sync_hot_reports"
        )

        self.scheduler.start()
        print("✅ 研报同步任务已启动")

    async def sync_hot_stocks(self):
        """同步热门股票研报"""
        # TODO: 从热门股票表获取
        hot_stocks = ["000001", "000002", "600036", "600519"]

        for stock_code in hot_stocks:
            await self.sync_single_stock(stock_code)
```

### 4. API 接口设计

```python
from fastapi import APIRouter, Depends, Query
from typing import List, Optional
from app.models.user import User
from app.core.auth import get_current_user
from app.services.research_report_service import ResearchReportService
from app.core.database import get_database

router = APIRouter(prefix="/api/research-reports", tags=["研报"])

@router.get("/{stock_code}")
async def get_stock_reports(
    stock_code: str,
    days: int = Query(90, ge=1, le=365, description="查询最近 N 天"),
    limit: int = Query(50, ge=1, le=200, description="返回条数"),
    current_user: User = Depends(get_current_user)
):
    """
    查询指定股票的研报列表

    - **stock_code**: 股票代码 (如 000001)
    - **days**: 查询天数 (默认 90 天)
    - **limit**: 返回条数 (默认 50 条)
    """
    db = get_database()
    service = ResearchReportService(db)

    reports = await service.get_reports_by_stock(
        stock_code=stock_code,
        days=days,
        limit=limit
    )

    return {
        "stock_code": stock_code,
        "total": len(reports),
        "reports": reports
    }

@router.get("/{stock_code}/consensus")
async def get_consensus_rating(
    stock_code: str,
    days: int = Query(90, ge=1, le=365, description="统计周期"),
    current_user: User = Depends(get_current_user)
):
    """
    获取市场共识评级

    返回:
    - 平均评级
    - 评级分布
    - 平均目标价
    - 最高/最低目标价
    """
    db = get_database()
    service = ResearchReportService(db)

    consensus = await service.get_consensus_rating(
        stock_code=stock_code,
        days=days
    )

    return consensus

@router.get("/institutions")
async def get_institutions(
    current_user: User = Depends(get_current_user)
):
    """列出所有研报机构"""
    db = get_database()

    institutions = await db.stock_research_reports.distinct("institution")

    return {
        "institutions": sorted(institutions),
        "total": len(institutions)
    }

@router.get("/analysts/top")
async def get_top_analysts(
    limit: int = Query(20, ge=1, le=100),
    current_user: User = Depends(get_current_user)
):
    """
    获取最活跃分析师排行榜

    按研报数量排序
    """
    db = get_database()

    pipeline = [
        {"$group": {
            "_id": {
                "institution": "$institution",
                "analyst": "$analyst"
            },
            "report_count": {"$sum": 1},
            "avg_target_price": {"$avg": "$target_price"}
        }},
        {"$sort": {"report_count": -1}},
        {"$limit": limit}
    ]

    cursor = db.stock_research_reports.aggregate(pipeline)
    results = await cursor.to_list(length=limit)

    return {
        "top_analysts": results,
        "total": len(results)
    }
```

```python
from app.worker.research_report_sync import ResearchReportSync
```

## ⚠️ 注意事项

### 数据质量

1. **去重机制**: 同一机构同一股票同一日期只保留一条研报
2. **数据验证**: 目标价必须为正数,评级必须在映射表中存在
3. **过期处理**: 目标价超过 90 天自动标记为 `is_valid=false`

### 性能优化

1. **索引优化**: 确保查询字段都有索引支持
2. **分页查询**: 大列表查询必须分页
3. **缓存策略**: 共识评级计算结果缓存 1 小时

### 法律合规

1. **数据来源**: 仅用于个人学习,不得商用
2. **版权声明**: 研报内容版权归机构所有
3. **免责条款**: 提示用户研报仅供参考,不构成投资建议
